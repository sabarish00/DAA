ALGORITHM Dijkstra(V, Edges, Source)
INPUT:
    V      - set of vertices
    Edges  - set of directed edges (u, v) with weight w(u,v) >= 0
    Source - start vertex
OUTPUT:
    Dist[v] - shortest distance from Source to v
    Pred[v] - predecessor of v on a shortest path
BEGIN
    FOR each vertex v IN V DO                  // initialisation
        Dist[v] <- INFINITY
        Pred[v] <- NIL
        Done[v] <- FALSE
    END FOR
    Dist[Source] <- 0
    FOR count <- 1 TO |V| DO
        u <- vertex with minimum Dist[u] among vertices with Done[u] = FALSE   // selection
        IF Dist[u] = INFINITY THEN
            BREAK                              // remaining vertices unreachable
        END IF
        Done[u] <- TRUE                        // Dist[u] is now final
        FOR each edge (u, v) with weight w DO
            IF Done[v] = FALSE AND Dist[u] + w < Dist[v] THEN    // relaxation
                Dist[v] <- Dist[u] + w         // distance update
                Pred[v] <- u                   // predecessor update
            END IF
        END FOR
    END FOR                                    // termination: all vertices done or unreachable
    RETURN Dist, Pred
END

ALGORITHM BellmanFord(V, Edges, Source)
INPUT: vertices V, directed edges (u, v, w) with any real weight w, Source
OUTPUT: Dist[], Pred[] or the message NEGATIVE CYCLE
BEGIN
    FOR each vertex v IN V DO
        Dist[v] <- INFINITY
        Pred[v] <- NIL
    END FOR
    Dist[Source] <- 0
    FOR pass <- 1 TO |V| - 1 DO
        Changed <- FALSE
        FOR each edge (u, v, w) IN Edges DO
            IF Dist[u] != INFINITY AND Dist[u] + w < Dist[v] THEN
                Dist[v] <- Dist[u] + w
                Pred[v] <- u
                Changed <- TRUE
            END IF
        END FOR
        IF Changed = FALSE THEN
            BREAK
        END IF
    END FOR
    FOR each edge (u, v, w) IN Edges DO        // |V|-th pass: cycle test
        IF Dist[u] != INFINITY AND Dist[u] + w < Dist[v] THEN
            RETURN NEGATIVE CYCLE
        END IF
    END FOR
    RETURN Dist, Pred
END
