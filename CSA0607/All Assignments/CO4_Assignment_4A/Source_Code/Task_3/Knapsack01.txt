ALGORITHM Knapsack01(ID[1..n], Weight[1..n], Value[1..n], Capacity)
INPUT:
    ID[i], Weight[i] > 0 (integer), Value[i] >= 0 for i = 1..n
    Capacity >= 0 (integer)
OUTPUT:
    BestValue - maximum total value
    Chosen    - list of IDs of the items selected
BEGIN
    // initialisation: row 0 and column 0 are the base states
    FOR c <- 0 TO Capacity DO
        DP[0][c] <- 0
    END FOR
    FOR i <- 0 TO n DO
        DP[i][0] <- 0
    END FOR
    // fill rows in increasing i: row i needs only row i-1
    FOR i <- 1 TO n DO
        FOR c <- 1 TO Capacity DO
            DP[i][c] <- DP[i-1][c]                        // exclude item i
            IF Weight[i] <= c THEN                        // item i may be included
                IncludeValue <- Value[i] + DP[i-1][c - Weight[i]]
                IF IncludeValue > DP[i][c] THEN
                    DP[i][c] <- IncludeValue
                END IF
            END IF
        END FOR
    END FOR
    BestValue <- DP[n][Capacity]
    // reconstruction: walk back from the final state
    Chosen <- empty list
    c <- Capacity
    FOR i <- n DOWNTO 1 DO
        IF DP[i][c] != DP[i-1][c] THEN                    // value changed => item i was taken
            APPEND ID[i] to Chosen
            c <- c - Weight[i]
        END IF
    END FOR
    REVERSE Chosen
    RETURN BestValue, Chosen
END
